uvicorn[standard]==0.30.1

# ─── MongoDB / ODM ───────────────────────────────────────────────────────────

# ─── Rate Limiting ────────────────────────────────────────────────────────────
slowapi==0.1.9

# ─── Validation / Settings ───────────────────────────────────────────────────
pydantic==2.8.2
pydantic-settings==2.4.0
python-multipart==0.0.9
# Pulled in by pydantic's EmailStr (models/user.py, schemas/auth_schema.py) —
# not imported directly anywhere, so easy to drop by accident when trimming
# "unused" imports. Without it, any request touching EmailStr raises at
# validation time with an import error, not a clean 422.
email-validator==2.3.0

# ─── Utilities ────────────────────────────────────────────────────────────────
python-dotenv==1.0.1

# ─── AI Assistant (Milestone 4) ────────────────────────────────────────────────
# google-generativeai: Gemini, the primary provider.
# openai: also used for the Groq fallback (Groq's API is OpenAI-compatible) —
# no OpenAI account/key involved for that path, just this client's request shape.
google-generativeai==0.8.3
openai==1.51.2

# ─── ML serving (goal-completion model) ────────────────────────────────────────
# services/goal_completion_service.py lazily `import joblib` to load
# models_store/goal_completion.joblib at request time — a real runtime path
# (GET /users/me/goals/predictions), not just a training-script concern.
# scikit-learn and numpy aren't imported directly there, but the pickled
# artifact is a fitted sklearn Pipeline (StandardScaler + LogisticRegression)
# built on numpy arrays — joblib.load needs both installed to unpickle it, or
# it fails at prediction time with an import error that pytest's mocked-DB
# suite will never see, because it never loads the real artifact.
joblib==1.4.2
scikit-learn==1.5.0
numpy==1.26.4
